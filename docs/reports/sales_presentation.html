# src/hr_agentic/knowledge/okf_rules.py (Section 22 Bereavement Rule)
IMMEDIATE_FAMILY = {"spouse", "partner", "child", "parent", "mother", "father", "sibling", "brother", "sister"}
EXTENDED_FAMILY = {"grandparent", "grandfather", "grandmother", "in-law"}

def evaluate_bereavement_leave(relationship: str) -> Dict[str, Any]:
    rel = relationship.lower().strip()
    if any(k in rel for k in ["dog", "cat", "pet", "animal"]):
        return {
            "eligible": False,
            "entitlement_days": 0,
            "policy_section": "Section 22.4",
            "message": "Bereavement leave is restricted to immediate and extended human family. Pets are not covered; please use annual vacation leave."
        }
    if any(k in rel for k in IMMEDIATE_FAMILY):
        return {"eligible": True, "entitlement_days": 5, "policy_section": "Section 22.2"}
    if any(k in rel for k in EXTENDED_FAMILY):
        return {"eligible": True, "entitlement_days": 2, "policy_section": "Section 22.3"}
    return {"eligible": False, "entitlement_days": 0, "policy_section": "Section 22.1"}
