def calculate_working_days(start_date: str, end_date: str) -> int:
    # Calculates working business days, strictly excluding weekends & holidays.
    start = datetime.strptime(start_date, "%Y-%m-%d").date()
    end = datetime.strptime(end_date, "%Y-%m-%d").date()
    if start > end:
        raise ValidationError("Start date cannot be after end date")
    
    working_days = 0
    cur = start
    while cur <= end:
        # Exclude Saturdays (5) and Sundays (6)
        if cur.weekday() < 5 and cur not in COMPANY_HOLIDAYS:
            working_days += 1
        cur += timedelta(days=1)
    return working_days
