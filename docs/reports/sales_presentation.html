# app/middleware/validation.py — Excludes holidays and weekends deterministically
def calculate_working_days(start_date: date, end_date: date) -> int:
    current = start_date
    working_days = 0
    while current <= end_date:
        if current.weekday() < 5 and current not in COMPANY_HOLIDAYS_2026:
            working_days += 1
        current += timedelta(days=1)
    return working_days
# app/middleware/validation.py — Demotes non-critical peripheral tickets per POL-IT-02
def enforce_incident_priority_anti_inflation(category: str, raw_priority: str) -> str:
    peripheral_keywords = ["mouse", "keyboard", "monitor", "headset", "dock"]
    if raw_priority in ["P1", "P2"] and any(kw in category.lower() for kw in peripheral_keywords):
        return "P4" # Auto-demote peripheral hardware to P4 Low
    return raw_priority
